class Solution {
public:
    vector<int> findMissing(vector<int>& a, vector<int>& b) {
        unordered_set<int> s(b.begin(), b.end());
        vector<int> ans;

        for (int x : a) {
            if (s.find(x) == s.end()) {
                ans.push_back(x);
            }
        }

        return ans;
    }
};
