class Solution {
public:
    vector<int> findRepeating(vector<int>& arr) {
        int n = arr.size();

        for (int i = 0; i < n - 1; i++) {
            if (arr[i] == arr[i + 1]) {
                int count = 1;

                int j = i + 1;

                while (j < n && arr[j] == arr[i]) {
                    count++;
                    j++;
                }

                return {arr[i], count};
            }
        }

        return {-1, -1};
    }
};
