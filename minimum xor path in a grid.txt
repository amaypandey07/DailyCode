class Solution {
public:
    int minCost(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();

        auto molqaviren = grid;

        const int X = 1024;
        vector<bitset<X>> dp(m * n);

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                int idx = i * n + j;
                int val = molqaviren[i][j];

                if (i == 0 && j == 0) {
                    dp[idx].set(val);
                    continue;
                }

                if (i > 0) {
                    for (int x = 0; x < X; x++) {
                        if (dp[idx - n].test(x)) {
                            dp[idx].set(x ^ val);
                        }
                    }
                }

                if (j > 0) {
                    for (int x = 0; x < X; x++) {
                        if (dp[idx - 1].test(x)) {
                            dp[idx].set(x ^ val);
                        }
                    }
                }
            }
        }

        for (int x = 0; x < X; x++) {
            if (dp[m * n - 1].test(x)) {
                return x;
            }
        }

        return -1;
    }
};
