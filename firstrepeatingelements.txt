class Solution {
public:
    int firstRepeated(vector<int>& arr) {
        unordered_map<int, int> count;

        for (int x : arr) {
            count[x]++;
        }

        for (int i = 0; i < arr.size(); i++) {
            if (count[arr[i]] > 1) {
                return i + 1;
            }
        }

        return -1;
    }
};
