class Solution {
public:
    vector<int> countLessEq(vector<int>& a, vector<int>& b) {
        sort(b.begin(), b.end());

        vector<int> ans;

        for (int i = 0; i < a.size(); i++) {
            int low = 0;
            int high = b.size() - 1;
            int count = 0;

            while (low <= high) {
                int mid = low + (high - low) / 2;

                if (b[mid] <= a[i]) {
                    count = mid + 1;
                    low = mid + 1;
                } else {
                    high = mid - 1;
                }
            }

            ans.push_back(count);
        }

        return ans;
    }
};
