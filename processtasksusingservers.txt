class Solution {
public:
    vector<int> assignTasks(vector<int>& servers, vector<int>& tasks) {
        using pii = pair<int, int>;
        using ti = tuple<long long, int, int>;

        priority_queue<pii, vector<pii>, greater<pii>> idle;

        priority_queue<ti, vector<ti>, greater<ti>> busy;

        int n = servers.size();
        int m = tasks.size();

        vector<int> ans(m);

        for (int i = 0; i < n; i++) {
            idle.push({servers[i], i});
        }

        for (int j = 0; j < m; j++) {

            while (!busy.empty() && get<0>(busy.top()) <= j) {
                auto [time, weight, index] = busy.top();
                busy.pop();

                idle.push({weight, index});
            }

            if (!idle.empty()) {
                auto [weight, index] = idle.top();
                idle.pop();

                ans[j] = index;

                busy.push({(long long)j + tasks[j], weight, index});
            }
            else {
                auto [time, weight, index] = busy.top();
                busy.pop();

                ans[j] = index;

                busy.push({time + tasks[j], weight, index});
            }
        }

        return ans;
    }
};
