class Solution {
public:
    long long minOperations(vector<int>& nums, int x, int k) {
        int n = nums.size();

        multiset<int> lo, hi;
        long long sumLo = 0, sumHi = 0;

        auto balance = [&]() {
            int target = (lo.size() + hi.size() + 1) / 2;

            while (lo.size() > target) {
                auto it = prev(lo.end());
                int v = *it;

                lo.erase(it);
                sumLo -= v;

                hi.insert(v);
                sumHi += v;
            }

            while (lo.size() < target) {
                auto it = hi.begin();
                int v = *it;

                hi.erase(it);
                sumHi -= v;

                lo.insert(v);
                sumLo += v;
            }
        };

        auto add = [&](int v) {
            if (lo.empty() || v <= *lo.rbegin()) {
                lo.insert(v);
                sumLo += v;
            } else {
                hi.insert(v);
                sumHi += v;
            }

            balance();
        };

        auto remove = [&](int v) {
            auto it = lo.find(v);

            if (it != lo.end()) {
                lo.erase(it);
                sumLo -= v;
            } else {
                auto it2 = hi.find(v);
                hi.erase(it2);
                sumHi -= v;
            }

            balance();
        };

        auto getCost = [&]() -> long long {
            long long median = *lo.rbegin();

            return median * (long long)lo.size() - sumLo
                 + sumHi - median * (long long)hi.size();
        };

        vector<long long> cost;

        for (int i = 0; i < n; i++) {
            add(nums[i]);

            if (i >= x) {
                remove(nums[i - x]);
            }

            if (i >= x - 1) {
                cost.push_back(getCost());
            }
        }

        const long long INF = LLONG_MAX / 4;

        vector<vector<long long>> dp(
            k + 1, vector<long long>(n + 1, INF)
        );

        for (int i = 0; i <= n; i++) {
            dp[0][i] = 0;
        }

        for (int j = 1; j <= k; j++) {
            for (int i = 1; i <= n; i++) {
                dp[j][i] = dp[j][i - 1];

                if (i >= x && dp[j - 1][i - x] != INF) {
                    dp[j][i] = min(
                        dp[j][i],
                        dp[j - 1][i - x] + cost[i - x]
                    );
                }
            }
        }

        return dp[k][n];
    }
};
